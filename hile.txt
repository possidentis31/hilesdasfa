    -- 2. FPS BOOST SİSTEMİ FONKSİYONU

    local function ApplyFPSBoost(state)
        Lighting.GlobalShadows = not state
        Lighting.FogEnd = state and 9e9 or 100000

        task.spawn(function()
            for _, v in pairs(workspace:GetDescendants()) do
                if not scriptRunning then break end
                if v:IsA("BasePart") then
                    v.CastShadow = not state
                    if state and (v:IsA("Part") or v:IsA("MeshPart")) then
                        v.Material = Enum.Material.SmoothPlastic
                    end
                elseif state and (v:IsA("Decal") or v:IsA("Texture")) then
                    v.Transparency = 1
                end
            end
        end)
    end

            TweenService:Create(UIKeys[input.KeyCode][2], TweenInfo.new(0), {Color = Color3.fromRGB(255, 50, 50)}):Play()
        end
    end)

    inputEndedCon2 = UserInputService.InputEnded:Connect(function(input, gp)
        if not scriptRunning then inputEndedCon2:Disconnect() return end
        if UIKeys[input.KeyCode] then
            TweenService:Create(UIKeys[input.KeyCode][1], TweenInfo.new(0.4, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {BackgroundColor3 = Color3.fromRGB(15, 10, 10)}):Play()
            TweenService:Create(UIKeys[input.KeyCode][2], TweenInfo.new(0.4, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Color = Color3.fromRGB(100, 100, 100)}):Play()
        end
    end)

    ---------------------------------------------------------
    -- 5. HEDEF BULMA & ANA DÖNGÜ (ESP + AIM + BHOP)
    ---------------------------------------------------------
    local function GetTarget()
        local target, maxDist = nil, 400
        for _, p in pairs(Players:GetPlayers()) do
            if p == LocalPlayer or R7Config.Whitelist[p.Name] or R7Config.Whitelist[p.DisplayName] then continue end
            local char = p.Character
            if char and char.Parent == workspace and char:FindFirstChild("Head") and char:FindFirstChild("Humanoid") and char.Humanoid.Health > 0.1 then
                local head = char.Head
                local pos, onScreen = Camera:WorldToScreenPoint(head.Position)
                if onScreen then
                    local canSee = true
                    if not R7Config.Wallbang then
                        local hit = workspace:Raycast(Camera.CFrame.Position, (head.Position - Camera.CFrame.Position).Unit * 1000)
                        if hit and hit.Instance and not hit.Instance:IsDescendantOf(char) then canSee = false end
                    end
                    if canSee then
                        local mouseDist = (Vector2.new(pos.X, pos.Y) - Vector2.new(Mouse.X, Mouse.Y)).Magnitude
                        if mouseDist < maxDist then maxDist = mouseDist; target = head end
                    end
                end
            end
        end
        return target
    end

    local renderSteppedCon
    renderSteppedCon = RunService.RenderStepped:Connect(function()
        if not scriptRunning then 
            renderSteppedCon:Disconnect() 
            return 
        end

        if Main.Visible then
            pcall(function() UserInputService.MouseIconEnabled = true; UserInputService.MouseBehavior = Enum.MouseBehavior.Default end)
        end

        for _, p in pairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") and p.Character:FindFirstChild("Humanoid") then
                local hrp = p.Character.HumanoidRootPart
                local hum = p.Character.Humanoid
                local espBox = hrp:FindFirstChild("R7ESP")
                
                if R7Config.ESP and hum.Health > 0 then
                    if not espBox then
                        espBox = Instance.new("BillboardGui", hrp)
                        espBox.Name = "R7ESP"
                        espBox.Size = UDim2.new(4, 0, 5.5, 0)
                        espBox.AlwaysOnTop = true
                        
                        local frame = Instance.new("Frame", espBox)
                        frame.Name = "BoxFrame"
                        frame.Size = UDim2.new(1, 0, 1, 0)
                        frame.BackgroundTransparency = 1
                        local stroke = Instance.new("UIStroke", frame)
                        stroke.Thickness = 2
                        
                        local nameLabel = Instance.new("TextLabel", espBox)
                        nameLabel.Name = "NameLabel"
                        nameLabel.Size = UDim2.new(2, 0, 0.2, 0) 
                        nameLabel.Position = UDim2.new(-0.5, 0, -0.25, 0)
                        nameLabel.BackgroundTransparency = 1
                        nameLabel.Font = Enum.Font.GothamBold
                        nameLabel.TextSize = 14 
                        local txtStroke = Instance.new("UIStroke", nameLabel)
                        txtStroke.Thickness = 1.5

                        local hbBg = Instance.new("Frame", espBox)
                        hbBg.Name = "HealthBg"
                        hbBg.Size = UDim2.new(0.08, 0, 1, 0)
                        hbBg.Position = UDim2.new(-0.12, 0, 0, 0)
                        hbBg.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
                        hbBg.BorderSizePixel = 0

                        local hbFill = Instance.new("Frame", hbBg)
                        hbFill.Name = "HealthFill"
                        hbFill.BorderSizePixel = 0
                    end
                    
                    local hpPercent = math.clamp(hum.Health / hum.MaxHealth, 0, 1)
                    espBox.HealthBg.HealthFill.Size = UDim2.new(1, 0, hpPercent, 0)
                    espBox.HealthBg.HealthFill.Position = UDim2.new(0, 0, 1 - hpPercent, 0)
                    espBox.HealthBg.HealthFill.BackgroundColor3 = Color3.new(1 - hpPercent, hpPercent, 0)

                    local isSafe = R7Config.Whitelist[p.Name] or R7Config.Whitelist[p.DisplayName]
                    local espColor = isSafe and Color3.fromRGB(50, 255, 50) or Color3.fromRGB(220, 20, 40)
                    
                    espBox.BoxFrame.UIStroke.Color = espColor
                    espBox.NameLabel.TextColor3 = espColor
                    espBox.NameLabel.Text = p.DisplayName .. " [" .. math.floor(hum.Health) .. "HP]"
                else
                    if espBox then espBox:Destroy() end
                end
            end
        end

        local currentTarget = nil
        if R7Config.Aimbot then
            currentTarget = GetTarget()
            if currentTarget then 
                local camPos = Camera.CFrame.Position
                local targPos = currentTarget.Position
                if (camPos - targPos).Magnitude > 0.1 then Camera.CFrame = CFrame.lookAt(camPos, targPos) end
            end
        end

        if R7Config.AutoShot then
            if currentTarget then
                if typeof(mouse1click) == "function" then mouse1click() end
            elseif Mouse.Target and Mouse.Target.Parent:FindFirstChild("Humanoid") then
                local pName = Mouse.Target.Parent.Name
                if not R7Config.Whitelist[pName] and typeof(mouse1click) == "function" then mouse1click() end
            end
        end

        if R7Config.FOVValue ~= 70 then
            Camera.FieldOfView = math.clamp(R7Config.FOVValue, 1, 120)
            if R7Config.FOVValue > 120 then
                LocalPlayer.DevCameraOcclusionMode = Enum.DevCameraOcclusionMode.Invisicam
                Camera.CFrame = Camera.CFrame * CFrame.new(0, 0, (R7Config.FOVValue - 120) / 10)
            else
                LocalPlayer.DevCameraOcclusionMode = Enum.DevCameraOcclusionMode.Zoom
            end
        else
            LocalPlayer.DevCameraOcclusionMode = Enum.DevCameraOcclusionMode.Zoom
        end
    end)

    ---------------------------------------------------------
    -- 6. HAREKET FİZİKLERİ (BHOP & NOCLIP)
    ---------------------------------------------------------
    local steppedConnection
    steppedConnection = RunService.Stepped:Connect(function()
        if not scriptRunning then 
            steppedConnection:Disconnect() 
            return 
        end
        if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid") then
            local char = LocalPlayer.Character
            local hum = char.Humanoid
            local hrp = char:FindFirstChild("HumanoidRootPart")

            if R7Config.Bhop and hrp and hum.Health > 0 then
                if UserInputService:IsKeyDown(Enum.KeyCode.Space) and hum.MoveDirection.Magnitude > 0 then
                    local speedBoost = 50 
                    local currentVel = hrp.Velocity
                    hrp.Velocity = Vector3.new(hum.MoveDirection.X * speedBoost, currentVel.Y, hum.MoveDirection.Z * speedBoost)
                    
                    if hum:GetState() ~= Enum.HumanoidStateType.Jumping and hum:GetState() ~= Enum.HumanoidStateType.Freefall then
                        hum:ChangeState(Enum.HumanoidStateType.Jumping)
                    end
                end
            end

            if R7Config.Noclip then
                for _, part in pairs(char:GetDescendants()) do
                    if part:IsA("BasePart") then part.CanCollide = false end
                end
            end
        end
    end)

    local jumpReqCon
    jumpReqCon = UserInputService.JumpRequest:Connect(function()
        if not scriptRunning then jumpReqCon:Disconnect() return end
        if R7Config.InfJump and LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid") then
            LocalPlayer.Character.Humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
        end
    end)

    ---------------------------------------------------------
    -- 7. TEMİZLEYEN KAPATMA FONKSİYONU (ANA MENÜ İÇİN)
    ---------------------------------------------------------
    CloseButton.MouseButton1Click:Connect(function()

        for _, p in pairs(Players:GetPlayers()) do

                local espBox = p.Character.HumanoidRootPart:FindFirstChild("R7ESP")
                if espBox then espBox:Destroy() end
            end
        end
        Camera.FieldOfView = 70
        LocalPlayer.DevCameraOcclusionMode = Enum.DevCameraOcclusionMode.Zoom
        ScreenGui:Destroy()
        FPSGui:Destroy()
    end)
end
